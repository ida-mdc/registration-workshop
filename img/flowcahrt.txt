digraph ImageRegistrationGuideline {
    rankdir=LR;
    nodesep=0.1;
    ranksep=0.1;
    node [shape=rectangle, style=rounded, fontsize=16, fontname="Helvetica"];
    
    size="9,9.7"; // Width=8 inches, Height=10 inches
    ratio=compress; // Adjust the ratio to fit the height

    // Preprocessing Decision
    preprocessing_check [label="Is \npreprocessing \nneeded?", shape=sqaure, style=filled, fillcolor=lightblue];

    // Preprocessing Steps
    preprocessing_steps [label="Correct \nIntensity, \nResize, \nor \nSegment.."];

    // Deep Learning Decision
    deep_learning_check [label="Do you \nhave the \nnecessary \nexpertise,\ncomputational \nresources,\nand perhaps\ntraining data\nfor\nDeep\nLearning?", shape=sqaure, style=filled, fillcolor=lightblue];

    // Deep Learning Registration
    deep_learning_registration [label="Apply \nDeep \Learning \nBased \nRegistration\n(e.g., \nVoxelMorph, \nDeepReg)", shape=ellipse, style=filled, fillcolor=lightgrey];

    // Step 1: Are the images very similar?
    similarity_check [label="Are the \nimages \nvery \nsimilar?\n(Minimal \ntransformation)", shape=sqaure, style=filled, fillcolor=lightblue];

    // Paths from similarity_check
    intensity_based [label="Use \nIntensity-Based \nRegistration\n(e.g., \nCross-Correlation)"];
    mutual_information [label="\nUse \nHistogram\nBased \nRegistration\n(e.g., \nMutual\nInformation)"];
    feature_based [label="Use \nFeature\nBased \nMethods"];
    retake_images [label="Retake \nImages \nwith \nExtrinsic \nLandmarks\nor \nImproved \nFeatures", shape=ellipse, style=filled, fillcolor=lightgrey];

    // Transformation Type Decision
    transformation_type [label="Select \nTransformation \nType", shape=sqaure, style=filled, fillcolor=lightblue];

    // Local Deformations Decision
    local_deformations [label="Are local \ndeformations \npresent?", shape=sqaure, style=filled, fillcolor=lightblue];

    // Transformation Methods
    rigid_transformation [label="Calculate \nand \nApply \nRigid, \nAffine, or \nPerspective \nTransformation"];
    elastic_transformation [label="Calculate \nand Apply \nElastic \nTransformation\n(e.g., \nB-Splines, \nTPS)"];

    // Final Step
    final_step [label="Apply \nSame \nTransformation \nto \nAdditional \nImages\nfrom \nSource \nSpace", shape=ellipse, style=filled, fillcolor=lightgrey];

    // Branching from Preprocessing Check
    preprocessing_check -> preprocessing_steps [label="Yes"];
    preprocessing_check -> deep_learning_check [label="No"];

    // Linking Preprocessing Steps to Deep Learning Check
    preprocessing_steps -> deep_learning_check;

    // Branching from Deep Learning Check
    deep_learning_check -> deep_learning_registration [label="Yes"];
    deep_learning_check -> similarity_check [label="No"];

    // Deep Learning Registration is now an endpoint
    // deep_learning_registration -> final_step; // Removed as per instructions

    // Branching from Similarity Check
    similarity_check -> intensity_based [label="Yes"];
    similarity_check -> mutual_information [label="Yes/\nSomehow"];
    similarity_check -> feature_based [label="Yes/\nSomehow/\nNo"];
    similarity_check -> retake_images [label="Not at all"];

    // Linking Registration Methods to Transformation Type
    intensity_based -> transformation_type;
    mutual_information -> transformation_type;

    // Split Feature-Based Methods
    feature_based -> classic_feature_based [label="Image Features"];
    classic_feature_based [label="Classic \nFeature\nBased \nMethods\n(e.g., \nSIFT, \nORB, \nSURF)"];
    classic_feature_based -> transformation_type;
    
    feature_based -> pose_estimation_based [label="Model Based"];
    pose_estimation_based [label="Model-Based\nFeature\nDetection\n(e.g., \nPose \nEstimation \nvia \nDeepLabCut)"];
    pose_estimation_based -> transformation_type;
    
    // Decision on Transformation Type
    transformation_type -> rigid_transformation [label="Global\nDeformations"];

    // Linking to Local Deformations Check
    rigid_transformation -> local_deformations;

    // Decision on Local Deformations
    local_deformations -> elastic_transformation [label="Yes"];
    local_deformations -> final_step [label="No"];

    // Elastic Transformation leads to Final Step
    elastic_transformation -> final_step;

    // Final Step from Rigid Transformation if No Elastic Needed
    rigid_transformation -> final_step [style=invis]; // Invisible edge to maintain graph structure

    // Retake Images is an endpoint
    retake_images -> final_step [style=invis]; // Invisible edge to maintain graph structure

    // Styling for Clarity
    edge [fontsize=12, fontname="Helvetica"];
    node [fontsize=12, fontname="Helvetica"];
}
